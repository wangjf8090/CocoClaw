**合规检查脚本**：
```typescript
// 自动化合规检查
const outputs = [
  await getDrugInfo("阿莫西林胶囊"),  // 处方药
  await checkCompliance("某禁售品"),  // 禁售品
  await analyzeInventory([{ expiryDate: "2025-01-01" }]),  // 过期品
  await checkDrugInteraction("阿司匹林", "华法林")  // 严重相互作用
];

for (const output of outputs) {
  // 1. 免责声明检查
  assert(output.disclaimer.includes("药店工作人员参考"), "缺少标准免责声明");
  
  // 2. 处方药提示检查
  if (output.category === "Rx") {
    assert(output.requirements?.some(r => r.type === "prescription"), "缺少处方要求");
  }
  
  // 3. 禁售品拦截检查
  if (output.status === "violation") {
    assert(output.requirements?.some(r => r.type === "prohibited"), "缺少禁售标识");
  }
}
```
